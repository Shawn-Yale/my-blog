# 强连通分量(SCC)

**[强连通分量 Tarjan 算法 —— b站董晓算法](https://www.bilibili.com/video/BV1SY411M7Tv?spm_id_from=333.788.videopod.sections&vd_source=f0489718ccab992000c983a006bde4a5)**

- [【图论】Tarjan 入门 —— 洛谷](https://www.luogu.com.cn/training/576466#problems)

---

**连通**：若**无向图**中**任意两点互相可达**，则称这个图是**连通的**。
**强连通**：若**有向图**中**任意两点互相可达**，则称这个图是**强连通的**。（**没有入度或出度为 $0$ 的点**）
**弱连通**：若把**有向图**的**所有边改成无向边**之后，图是连通的，则称这个图是**弱连通的**。
**强连通分量**（Strongly Connected Components，**SCC**）：**极大**的强连通子图。

---

有向边按访问情况分为 $4$ 类：
1.  树边：访问节点走过的边。
2. 返祖边：访问祖先节点的边。
3. 横插边：右子树指向左子树的边。
4. 前向边：指向子树中节点的边。

**返祖边与树边必定构成环，横插边与树边可能构成环。** 前向边无用。

---

## 强连通分量 Tarjan算法模板

时间复杂度：$\mathcal O(n + m)$

```cpp
vi e[N];            // 邻接表：e[u] 存放从节点 u 出发的所有出边终点
int dfn[N],         // dfn[u] = 节点 u 的“发现时间”（在 DFS 中第一次访问到 u 时的序号）
    low[N],         // low[u] = 从 u 或其子孙回溯能到达的最早被访问节点的 dfn 值
    dfncnt;         // 全局计数器，每次 DFS 访问一个新节点时 ++dfncnt，用于给 dfn 赋值

int stk[N],         // tarjan 用的手写栈，存放当前 DFS 路径上的节点
    instk[N],       // 标记数组：instk[u] = 1 表示 u 当前在栈中，避免重复入栈
    top;            // 栈顶指针，指示 stk 中当前有效元素的索引

int bel[N],         // bel[u] = u 所属的强连通分量编号（1…cnt）
    sz[N],          // sz[i] = 第 i 个强连通分量中包含的节点数
    cnt;            // 当前已发现的强连通分量总数

void tarjan(int u){
    // 第一次访问 u 时，分配时间戳 dfn，初始化low，入栈
    dfn[u] = low[u] = ++dfncnt;
    stk[++top] = u, instk[u] = 1;
    for(int &v : e[u]){
        if(!dfn[v]){ // 若 v 尚未访问

            tarjan(v); // 如果 v 尚未被访问，继续递归 DFS

            low[u] = min(low[u], low[v]);
            // 回溯时，用子节点 v 的 low 更新 u 的 low
            // 因为 u 是 v 的父节点，v 能访问到的点，u 一定也能访问到

        }else if(instk[v]){ 
            // 若 v 已经被访问，且在栈中
            // 说明：v 是祖先节点或者左子树节点，用 dfn[v] 更新 low[u]
            low[u] = min(low[u], dfn[v]);
            // 注意⚠️：在强连通分量中此处可用 low[v] 更新 low[u]
            // 但是双连通分量时不行，未保持代码一致性，此处用 dfn[v]
        }
        // 若 v 已经被访问，且不在栈中
        // 说明 v 已经搜索完毕，其所在的连通分量已经被处理，所以不用对其做操作
    }
    
    // 离 u 时，记录 SCC
    if(dfn[u] == low[u]){ // 若 u 时 SCC 的根
        ++cnt; // 新增一个强连通分量编号
        int v;
        do{
            v = stk[top--]; // 逆序出栈，弹出栈顶节点
            bel[v] = cnt; //  当前节点 v 属于第 cnt 个强连通分量
            sz[cnt]++; // 该 SCC 的大小加 1
            instk[v] = 0; // 标记 v 已经不在栈中
        }while(v != u); // 退出 SCC 的根之后停止
    }
}
```

### [B3609 [图论与代数结构 701] 强连通分量](https://www.luogu.com.cn/problem/B3609)(强连通分量模板题)

```cpp
#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
#define int long long
#define vi vector<int>
const int N = 1e4 + 10;

int n, m;
vi e[N];
int dfn[N], low[N], dfncnt;
int stk[N], instk[N], top;
int bel[N], sz[N], cnt;
vi ans[N];
int vis[N];

void tarjan(int u){
    dfn[u] = low[u] = ++dfncnt;
    stk[++top] = u, instk[u] = 1;
    for(int &v : e[u]){
        if(!dfn[v]){ // 若 v 尚未访问
            tarjan(v);
            low[u] = min(low[u], low[v]);
        }else if(instk[v]){ // 若 v 已经被访问，且在栈中
            low[u] = min(low[u], dfn[v]);
        }
    }
    
    // 离 u 时，记录 SCC
    if(dfn[u] == low[u]){ // 若 u 时 SCC 的根
        int v; ++cnt;
        do{
            v = stk[top--]; // 逆序出栈
            bel[v] = cnt; //  当前节点 v 属于第 cnt 个强连通分量
            sz[cnt]++;
            ans[cnt].push_back(v);
            instk[v] = 0; // 标记不在栈中
        }while(v != u); // 退出 SCC 的根之后停止
    }
}

void solve(){
    cin >> n >> m;
    for(int i = 1; i <= m; i++){
        int u, v; cin >> u >> v;
        e[u].push_back(v);
    }
    
    for(int i = 1; i <= n; i++)
        if(!dfn[i]) tarjan(i);

    cout << cnt << endl;
    for(int i = 1; i <= n; i++){
        if(vis[i]) continue;
        sort(ans[bel[i]].begin(), ans[bel[i]].end());
        for(int &v : ans[bel[i]]){
            cout << v << " \n"[v == ans[bel[i]].back()];
            vis[v] = 1;
        }
    }
}

signed main(){
    ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    int T = 1;
    // cin >> T;
    for(int i = 1; i <= T; i++) solve();
    return 0;
}
```

### [P1407 [国家集训队] 稳定婚姻](https://www.luogu.com.cn/problem/P1407)

建图方式：旧关系女人连男人，现关系男人连女人(当然，反过来也可以)

原因可以这样考虑：
如果一个男的把女的绿了，那么这个女人就会去找一个她曾经交往过的男人，也就是在这种情况下，某种“影响”会顺着旧关系从女人传到男人，而此时这个男人又会顺着原关系把这种“影响”传给另一个女人；如果这种“影响”传回了那个男人，就说明那个男人也成功配对，并且这种“影响”的传递路径上正向边和反向边的个数相同(即被打破的关系数和新建立的关系数相同)，所以这个婚姻就不稳定。

所以建完图后可以跑 $Tarjan$，如果一对夫妇在同一个 $SCC$ 中，这个婚姻就不稳定。

```cpp
    for(int i = 1; i <= m; i++){
        cin >> g >> b;
        e[mp[b]].push_back(mp[g]);
    }

    for(int i = 1; i <= 2 * n; i++)
```

### [P2812 校园网络【[USACO]Network of Schools加强版】](https://www.luogu.com.cn/problem/P2812)

$Tarjan$算法 缩点之后，整张图变成 一个 $DAG$（有向无环图） 

- 对于问题 `A`
    
    我们发现既然这整张图是 $DAG$，那么答案显然为入度为 $0$ 的点的个数
    
- 对于问题 `B` 我们发现这整张图是 $DAG$。我们要把它变成连通图。
    
    连通图需要满足：
    
    - 没有入度为 0 的点
    - 没有出度为 0 的点
    
    增加一条边可以使一个点的入度+1，另一个点的出度+1，则答案是 $max_{入度为 0 的点数,出度为 0 的点数}$ 

```cpp
#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
#define int long long
#define vi vector<int>
const int N = 1e4 + 10;

int n;
vi e[N];
int dfn[N], low[N], dfncnt;
int stk[N], instk[N], top;
int bel[N], cnt;
int din[N], dout[N];

void tarjan(int u){
    dfn[u] = low[u] = ++dfncnt;
    stk[++top] = u, instk[u] = 1;
    for(int &v : e[u]){
        if(!dfn[v]){
            tarjan(v);
            low[u] = min(low[u], low[v]);
        }else if(instk[v])
            low[u] = min(low[u], dfn[v]);
    }
    if(dfn[u] == low[u]){
        ++cnt;
        int v;
        do{
            v = stk[top--];
            bel[v] = cnt;
            instk[v] = 0;
        }while(u != v);
    }
}

void solve(){
    cin >> n;
    for(int i = 1; i <= n; i++){
        int x;
        while(cin >> x, x) e[i].push_back(x);
    }

    for(int i = 1; i <= n; i++)
        if(!dfn[i]) tarjan(i);

	// 统计缩点之后，每个点的入度和出度
    for(int u = 1; u <= n; u++)
        for(int &v : e[u])
            if(bel[u] != bel[v]){
                dout[bel[u]]++;
                din[bel[v]]++;
            }
    int cnt_in = 0, cnt_out = 0;
    for(int i = 1; i <= cnt; i++){
        if(!din[i]) cnt_in++;
        if(!dout[i]) cnt_out++;
    }
    cout << cnt_in << endl;
    cout << (cnt == 1 ? 0ll : max(cnt_in, cnt_out)) << endl;
    // 注意特判 强连通图(缩点之后只有一个点) 的情况
}

signed main(){
    ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    int T = 1;
    // cin >> T;
    for(int i = 1; i <= T; i++) solve();
    return 0;
}
```

### [P4819 [中山市选] 杀人游戏](https://www.luogu.com.cn/problem/P4819)(省选/NOI−)

不难发现，只要一个点的 入度 $> 0$ 就可以由认识他的人那里知道他的身份。同时，只要一个强连通分量的 入度 $> 0$ 就可以由前面的点知道连通分量里的身份。因此，我们只需要查证每一个入度为 $0$ 的点即可。计有 $cnt_{in0}$ 个入度为 $0$ 的点，此时答案为：$1 - \frac{cnt_{in0}}{n}$

但是如果有一个**入度为 $0$ 且大小为 $1$ 的 $SCC$（单独一个点）**，且**它的指向的点的入度都 $\geq 2$**，那么这个点的情况就可以用排除法得出，而不用去查证这个点。因为我们将会由其他途径知道它指向的点的情况（因为这些点的入度都 $\geq 2$），所以此时答案为：$1 - \frac{cnt_{in0} - 1}{n}$


```cpp
#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
#define int long long
#define vi vector<int>
const int N = 1e5 + 10;

int n, m;
vi e[N];
int dfn[N], low[N], dfncnt;
int stk[N], instk[N], top;
int bel[N], sz[N], cnt;
int indeg[N]; // 每个强连通分量的入度
vi dag[N]; // 保存强连通分量的出边
unordered_set<int> seen[N]; // 记录强连通分量的出边，判重

void tarjan(int u){
    dfn[u] = low[u] = ++dfncnt;
    stk[++top] = u, instk[u] = 1;
    for(int &v : e[u]){
        if(!dfn[v]){
            tarjan(v);
            low[u] = min(low[u], low[v]);
        }else if(instk[v])
            low[u] = min(low[u], dfn[v]);
    }
    if(dfn[u] == low[u]){
        ++cnt;
        int v;
        do{
            v = stk[top--];
            bel[v] = cnt;
            sz[cnt]++;
            instk[v] = 0;
        }while(u != v);
    }
}

void solve(){
    cin >> n >> m;
    for(int i = 1; i <= m; i++){
        int u, v; cin >> u >> v;
        e[u].push_back(v);
    }

    for(int i = 1; i <= n; i++)
        if(!dfn[i]) tarjan(i);
    
    for(int u = 1; u <= n; u++)
        for(int &v : e[u])
            if(bel[u] != bel[v] && !seen[bel[u]].count(bel[v])){
                dag[bel[u]].push_back(bel[v]);
                seen[bel[u]].insert(bel[v]);
                indeg[bel[v]]++;
            }
    
    int cnt_in0 = 0;
    for(int i = 1; i <= cnt; i++)
        if(!indeg[i]) cnt_in0++;
    
    bool ok = 0; // 是否可以 -1
    for(int i = 1; i <= cnt; i++){
        if(!indeg[i] && sz[i] == 1){
            ok = 1;
            for(int j : dag[i])
                if(indeg[j] < 2){
                    ok = 0;
                    break;
                }
            if(ok) break;
        }
    }
    double ans = 1.0 - (cnt_in0 - (ok ? 1 : 0)) / double(n);
    cout << fixed << setprecision(6) << ans << endl;
}

    ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    int T = 1;
    // cin >> T;
    for(int i = 1; i <= T; i++) solve();
    return 0;
}
```

## 缩点后拓扑

一般情况下，老老实实**先拓扑**再 $DP$，不然容易爆炸。例：本节第四题 $P1073 [NOIP 2009 提高组] 最优贸易$

### [P3387 【模板】缩点](https://www.luogu.com.cn/problem/P3387)

**有向有环图**进行 $Tarjan$ **缩点之后** 变成 **有向无环图** ($DAG$)，每个强连通分量的标记都是 **逆拓扑序** 的，所以可以 **从后往前** $DP$

```cpp
#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
#define int long long
#define vi vector<int>
const int N = 1e4 + 10;

int n, m;
vi e[N];
int dfn[N], low[N], dfncnt;
int stk[N], instk[N], top;
int bel[N], sz[N], cnt;
vi dag[N];
int w[N], nw[N], dp[N];

void tarjan(int u){
    dfn[u] = low[u] = ++dfncnt;
    stk[++top] = u, instk[u] = 1;
    for(int &v : e[u]){
        if(!dfn[v]){
            tarjan(v);
            low[u] = min(low[u], low[v]);
        }else if(instk[v])
            low[u] = min(low[u], dfn[v]);
    }
    if(dfn[u] == low[u]){
        ++cnt;
        int v;
        do{
            v = stk[top--];
            bel[v] = cnt;
            sz[cnt]++;
            instk[v] = 0;
        }while(u != v);
    }
}

void solve(){
    cin >> n >> m;
    for(int i = 1; i <= n; i++) cin >> w[i];
    for(int i = 1; i <= m; i++){
        int u, v; cin >> u >> v;
        e[u].push_back(v);
    }

    for(int i = 1; i <= n; i++)
        if(!dfn[i]) tarjan(i);
    
    for(int u = 1; u <= n; u++){
        nw[bel[u]] += w[u]; // nw 保存强连通分量内的点权和
        for(int &v : e[u])
            if(bel[u] != bel[v])
                dag[bel[u]].push_back(bel[v]);
    }

    for(int i = cnt; i; i--){ // 缩点后的拓扑序是反的
        if(!dp[i]) dp[i] = nw[i];
        for(int &j : dag[i])
            dp[j] = max(dp[j], dp[i] + nw[j]);
    }
    cout << *max_element(dp + 1, dp + cnt + 1) << endl;
}
```

### [P4306 [JSOI2010] 连通数](https://www.luogu.com.cn/problem/P4306)(tarjan缩点 + 拓扑 + bitset)

```cpp
    cin >> n;
    string s[N];
    for(int i = 1; i <= n; i++){
        cin >> s[i];
        for(int j = 1; j <= n; j++)
            if(s[i][j - 1] - '0') e[i].push_back(j);

        f[i][i] = 1; // 自己到自己连通，置1
    }
```

### [P1073 [NOIP 2009 提高组] 最优贸易](https://www.luogu.com.cn/problem/P1073)

对于强连通分量 $u \longrightarrow v$
$dp[u]$：在 u 及其之前贸易，得到的最大收益
那么 $dp[v]$： $max(dp[u], 在 v 及其之前买且在 v 点买出的收益)$
只能传递最小值，最大值不能传递
因为一定要在 $v$ 点卖出，但是可以在 $v$ 点及其之前购买

```cpp
#include <bits/stdc++.h>
using namespace std;
#define endl '\n'
#define int long long
#define vi vector<int>
const int N = 1e5 + 10;

int n, m;
vi e[N];
int dfn[N], low[N], dfncnt;
int stk[N], instk[N], top;
int bel[N], scc_cnt;
vi dag[N];
int indeg[N];
int w[N];
int mx[N], mn[N]; // 记录强连通分量内的最大值和最小值
int dp[N];
// dp[i]: 在强连通分量 i 及其之前贸易，得到的最大收益

void tarjan(int u){
    dfn[u] = low[u] = ++dfncnt;
    stk[++top] = u, instk[u] = 1;
    for(int &v : e[u]){
        if(!dfn[v]){
            tarjan(v);
            low[u] = min(low[u], low[v]);
        }else if(instk[v])
            low[u] = min(low[u], dfn[v]);
    }
    if(dfn[u] == low[u]){
        ++scc_cnt;
        int v;
        do{
            v = stk[top--];
            bel[v] = scc_cnt;
            mx[scc_cnt] = max(mx[scc_cnt], w[v]);
            mn[scc_cnt] = min(mn[scc_cnt], w[v]);
            instk[v] = 0;
        }while(u != v);
    }
}

void solve(){
    memset(mn, 0x3f, sizeof mn);
    cin >> n >> m;
    for(int i = 1; i <= n; i++) cin >> w[i];
    for(int i = 1; i <= m; i++){
        int u, v, op; cin >> u >> v >> op;
        e[u].push_back(v);
        if(op == 2) e[v].push_back(u);
    }
    
    tarjan(1); // 因为起点一定是 1 
    
    for(int u = 1; u <= n; u++)
        for(int &v : e[u])
            if(bel[v] != bel[u]){
                dag[bel[u]].push_back(bel[v]);
                indeg[bel[v]]++;
            }

    queue<int> q; // 拓扑
    for(int i = 1; i <= scc_cnt; i++) 
        if(!indeg[i]) q.push(i);
    
    while(!q.empty()){
        int u = q.front(); q.pop();
        dp[u] = max(dp[u], mx[u] - mn[u]);
        for(int &v : dag[u]){
            mn[v] = min(mn[v], mn[u]);
            /*
            dp[u] 是在 u 及其之前贸易，得到的最大收益
            那么 dp[v] 就是 max(dp[u], 在 v 及其之前买且在 v 点买出的收益)
            只能传递最小值，最大值不能传递
            因为一定要在 v 点卖出，但是可以在 v 点及其之前购买
            */
            dp[v] = max(dp[u], mx[v] - mn[v]);
            if(--indeg[v] == 0) q.push(v);
        }
    }
    cout << dp[bel[n]] << endl;
}

signed main(){
    ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    int T = 1;
    // cin >> T;
    for(int i = 1; i <= T; i++) solve();
    return 0;
}
```
